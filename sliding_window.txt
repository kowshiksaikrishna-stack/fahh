
#include <stdio.h>

int main()
{
    int n, w, i, j;

    printf("Enter the number of frames: ");
    scanf("%d", &n);

    printf("Enter the window size: ");
    scanf("%d", &w);

    int ack[100] = {0};

    printf("\nSelective Repeat Sliding Window Protocol\n");

    for(i = 0; i < n; i += w)
    {
        printf("\nSending frames:\n");

        for(j = i; j < i + w && j < n; j++)
        {
            printf("Frame %d sent\n", j);
        }

        printf("\nEnter ACK (1=Received, 0=Lost):\n");

        for(j = i; j < i + w && j < n; j++)
        {
            printf("ACK for Frame %d: ", j);
            scanf("%d", &ack[j]);
        }

        printf("\nRetransmitting lost frames:\n");

        for(j = i; j < i + w && j < n; j++)
        {
            if(ack[j] == 0)
            {
                printf("Frame %d retransmitted\n", j);
                printf("ACK received for Frame %d\n", j);
            }
        }
    }

    printf("\nAll frames transmitted successfully!\n");

    return 0;
}
